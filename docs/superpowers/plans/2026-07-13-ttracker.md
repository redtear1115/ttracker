# ttracker 台南垃圾車追蹤 PWA 實作計畫

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 純自用的台南市垃圾車 PWA——最愛清運點的下一班倒數、今日回收標記、即時車輛位置。

**Architecture:** Svelte 5 + Vite + TS 靜態 PWA；清運點資料 build 時打包（`scripts/fetch-points.ts` 產出 `data/points.json`）；即時 GPS 走 Cloudflare Pages Function `/api/gps` 代理（上游要求特定 Referer）；最愛存 localStorage。詳見 `docs/superpowers/specs/2026-07-13-tainan-garbage-truck-pwa-design.md`。

**Tech Stack:** Svelte 5（runes）、Vite 6、TypeScript strict、Vitest、vite-plugin-pwa、Leaflet + OpenStreetMap、Cloudflare Pages Functions。

- Node 20+、npm。所有指令在 repo root（`/Users/ray-lee/Projects/js/ttracker`）執行。
- TypeScript strict；純邏輯（日期、解析、地理）一律獨立於 UI，放 `src/lib/`，全部進 Vitest。

```ts
        try {

            method: 'POST',
            headers: {
              'Content-Type': 'application/json; charset=utf-8',
              Referer: 'https://clean.tnepb.gov.tw/GarbageTruckLiveFeed',
            },
            body: '{}',

          res.end(text)
        } catch {
          res.statusCode = 502
          res.end('{"error":"upstream"}')
        }
      })
    },
  }
}

  plugins: [svelte(), gpsDevProxy()],
```

```ts
const DAY_MAP: Record<string, number> = { 一: 1, 二: 2, 三: 3, 四: 4, 五: 5, 六: 6, 日: 7 }

export function parseDays(s: string | null): number[] {
  if (s == null || s === '' || s === '無清運') return []
  return s.split('、').map((part) => {
    const d = DAY_MAP[part.trim()]
    if (d === undefined) throw new Error(`無法解析星期值: ${JSON.stringify(s)}`)
    return d
  })
}
```

```ts
const NAMED: Record<string, string> = { quot: '"', amp: '&', lt: '<', gt: '>', apos: "'", '#39': "'" }

export function decodeEntities(s: string): string {
  return s.replace(/&(#x?[0-9a-fA-F]+|[a-zA-Z]+|#39);/g, (m, code: string) => {
    if (code.startsWith('#x') || code.startsWith('#X')) return String.fromCodePoint(parseInt(code.slice(2), 16))
    if (code.startsWith('#')) return String.fromCodePoint(parseInt(code.slice(1), 10))
    return NAMED[code] ?? m
  })
}

/** 在 unescape 後的文字中，從第一個 `"AREA"` 往回找 `[`，做字串感知的括號配對，抽出 JSON 陣列 */
export function extractJsonArray(html: string): Record<string, unknown>[] {
  const text = decodeEntities(html)
  const anchor = text.indexOf('"AREA"')
  if (anchor === -1) throw new Error('頁面中找不到 "AREA" 錨點——上游格式可能已改版')
  const start = text.lastIndexOf('[', anchor)
  let depth = 0
  let inString = false
  for (let i = start; i < text.length; i++) {
    const c = text[i]
    if (inString) {
      if (c === '\\') i++
      else if (c === '"') inString = false
    } else if (c === '"') inString = true
    else if (c === '[' || c === '{') depth++
    else if (c === ']' || c === '}') {
      depth--
      if (depth === 0) return JSON.parse(text.slice(start, i + 1))
    }
  }
  throw new Error('JSON 陣列未閉合——下載可能被截斷，請重試')
}
```

```ts
import { writeFileSync, mkdirSync } from 'node:fs'
import { extractJsonArray } from './extractJson'
import { parseDays } from '../src/lib/parseDays'
import type { CollectionPoint, PointsFile } from '../src/lib/types'

const SOURCE = 'https://data.tainan.gov.tw/Resource/84df8cd6-8741-41ed-919c-5105a28ecd6d?handler=GoJson'

interface RawPoint {
  AREA: string; ROUTEID: string; ROUTEORDER: number; VILLAGE: string | null
  POINTNAME: string; TIME: string | null; LONGITUDE: number | null
  LATITUDE: number | null; WORKDAY: string | null; RECYCLEDAY: string | null
}

async function main() {
  console.log('下載中（~28MB，需一兩分鐘）…')
  const res = await fetch(SOURCE)
  if (!res.ok) throw new Error(`下載失敗: HTTP ${res.status}`)
  const html = await res.text()
  const raw = extractJsonArray(html) as unknown as RawPoint[]
  console.log(`原始筆數: ${raw.length}`)
  if (raw.length < 10000) throw new Error('筆數異常偏低，疑似上游改版或截斷')

  const seen = new Map<string, number>()
  let skippedNoCoord = 0
  const points: CollectionPoint[] = []
  for (const r of raw) {
    if (r.LONGITUDE == null || r.LATITUDE == null) { skippedNoCoord++; continue }
    if (r.LONGITUDE < 119.9 || r.LONGITUDE > 120.7 || r.LATITUDE < 22.7 || r.LATITUDE > 23.5)
      throw new Error(`座標超出台南範圍: ${JSON.stringify(r)}`)
    if (r.TIME != null && !/^\d{2}:\d{2}$/.test(r.TIME))
      throw new Error(`TIME 格式異常: ${JSON.stringify(r.TIME)}`)
    const base = `${r.AREA}-${r.ROUTEID}-${r.ROUTEORDER}`
    const n = (seen.get(base) ?? 0) + 1
    seen.set(base, n)
    points.push({
      id: n === 1 ? base : `${base}-${n}`,
      area: r.AREA,
      routeId: r.ROUTEID,
      routeOrder: r.ROUTEORDER,
      village: r.VILLAGE ?? '',
      name: r.POINTNAME,
      time: r.TIME,
      lat: Math.round(r.LATITUDE * 1e6) / 1e6,
      lng: Math.round(r.LONGITUDE * 1e6) / 1e6,
      workDays: parseDays(r.WORKDAY),
      recycleDays: parseDays(r.RECYCLEDAY),
    })
  }
  const out: PointsFile = { updatedAt: new Date().toISOString().slice(0, 10), points }
  mkdirSync('data', { recursive: true })
  writeFileSync('data/points.json', JSON.stringify(out))
  console.log(`完成: ${points.length} 點（略過 ${skippedNoCoord} 筆缺座標）→ data/points.json`)
}

main()
```

```ts
export function isoDay(d: Date): number {
  return ((d.getDay() + 6) % 7) + 1
}

export function nextArrival(time: string | null, workDays: number[], now: Date): Date | null {
  if (!time || workDays.length === 0) return null
  const [h, m] = time.split(':').map(Number)
  for (let add = 0; add < 8; add++) {
    const d = new Date(now)
    d.setDate(d.getDate() + add)
    d.setHours(h, m, 0, 0)
    if (workDays.includes(isoDay(d)) && d.getTime() >= now.getTime()) return d
  }
  return null
}

export function minutesUntil(target: Date, now: Date): number {
  return Math.floor((target.getTime() - now.getTime()) / 60000)
}

export function isRecycleDay(recycleDays: number[], now: Date): boolean {
  return recycleDays.includes(isoDay(now))
}
```

```ts
export function haversineMeters(lat1: number, lng1: number, lat2: number, lng2: number): number {
  const R = 6371000
  const toRad = (x: number) => (x * Math.PI) / 180
  const dLat = toRad(lat2 - lat1)
  const dLng = toRad(lng2 - lng1)
  const a = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) * Math.sin(dLng / 2) ** 2
  return 2 * R * Math.asin(Math.sqrt(a))
}
```

```ts
import { haversineMeters } from './geo'
import type { CollectionPoint } from './types'

export interface Truck {
  plate: string; caption: string; at: Date; lat: number; lng: number
  direct: string; cartype: string; carId: string; substitutePlate: string
}

interface RawCar {
  car_licence: string; caption: string; dt: string; x: string; y: string
  direct: string; status: string; car_no: string; rcar_licence: string; cartype: string; car_id: string
}

export function parseCarsResponse(raw: string): Truck[] {
  const outer = JSON.parse(raw) as { d: string }
  const inner = JSON.parse(outer.d) as { DATA?: RawCar[] }
  return (inner.DATA ?? [])
    .map((c): Truck | null => {
      const lat = Number(c.y)
      const lng = Number(c.x)
      const at = new Date(c.dt.replace(' ', 'T'))
      if (!Number.isFinite(lat) || !Number.isFinite(lng) || Number.isNaN(at.getTime())) return null
      return {
        plate: c.car_licence, caption: c.caption, at, lat, lng,
        direct: c.direct, cartype: c.cartype, carId: c.car_id, substitutePlate: c.rcar_licence,
      }
    })
    .filter((t): t is Truck => t !== null)
}

export function isStale(truck: Truck, now: Date, maxAgeMs = 10 * 60_000): boolean {
  return now.getTime() - truck.at.getTime() > maxAgeMs
}

export interface TruckMatch {
  truck: Truck; nearestOrder: number; stopsAway: number; distanceM: number
}

const MATCH_RADIUS_M = 500

export function matchTruckToPoint(
  point: CollectionPoint, routePoints: CollectionPoint[], trucks: Truck[], now: Date,
): TruckMatch | null {
  let best: TruckMatch | null = null
  for (const truck of trucks) {
    if (isStale(truck, now)) continue
    let nearest: { order: number; dist: number } | null = null
    for (const p of routePoints) {
      const dist = haversineMeters(truck.lat, truck.lng, p.lat, p.lng)
      if (nearest === null || dist < nearest.dist) nearest = { order: p.routeOrder, dist }
    }
    if (!nearest || nearest.dist > MATCH_RADIUS_M) continue
    if (best === null || nearest.dist < best.distanceM) {
      best = { truck, nearestOrder: nearest.order, stopsAway: point.routeOrder - nearest.order, distanceM: nearest.dist }
    }
  }
  return best
}
```

```ts
import { parseCarsResponse, type Truck } from './trucks'

const INTERVAL_MS = 15_000

class GpsPoller {
  trucks = $state<Truck[]>([])
  status = $state<'idle' | 'ok' | 'error'>('idle')
  lastFetched = $state<Date | null>(null)
  #timer: ReturnType<typeof setInterval> | null = null

  async #fetch(): Promise<void> {
    try {
      const r = await fetch('/api/gps')
      if (!r.ok) throw new Error(String(r.status))
      this.trucks = parseCarsResponse(await r.text())
      this.status = 'ok'
      this.lastFetched = new Date()
    } catch {
      this.status = 'error'
    }
  }

  start(): void {
    if (this.#timer) return
    void this.#fetch()
    this.#timer = setInterval(() => void this.#fetch(), INTERVAL_MS)
    document.addEventListener('visibilitychange', this.#onVisibility)
  }

  stop(): void {
    if (this.#timer) clearInterval(this.#timer)
    this.#timer = null
    document.removeEventListener('visibilitychange', this.#onVisibility)
  }

  #onVisibility = (): void => {
    if (document.hidden) {
      if (this.#timer) clearInterval(this.#timer)
      this.#timer = null
    } else if (!this.#timer) {
      void this.#fetch()
      this.#timer = setInterval(() => void this.#fetch(), INTERVAL_MS)
    }
  }
}

export const gps = new GpsPoller()
```

- Produces: 使用者可（a）行政區 → 路線 → 清運點逐層瀏覽（b）「附近的點」geolocation 找最近 20 點（c）關鍵字過濾點名；每列有 ★ toggle。

- [ ] **Step 1: 實作 `SearchView.svelte`**

```svelte
<script lang="ts">
  import { areas, routesInArea, routePoints, allPoints } from '../lib/points'
  import { favorites } from '../lib/favorites.svelte'
  import { haversineMeters } from '../lib/geo'
  import type { CollectionPoint } from '../lib/types'

  let area = $state('')
  let routeKeySel = $state('')
  let keyword = $state('')
  let nearby = $state<CollectionPoint[] | null>(null)
  let geoError = $state('')

  const routes = $derived(area ? routesInArea(area) : [])
  const points = $derived.by(() => {
    if (nearby) return nearby
    if (!routeKeySel) return []
    const list = routePoints(routeKeySel)
    return keyword ? list.filter((p) => p.name.includes(keyword) || p.village.includes(keyword)) : list
  })

  function findNearby(): void {
    geoError = ''
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        nearby = [...allPoints]
          .map((p) => ({ p, d: haversineMeters(pos.coords.latitude, pos.coords.longitude, p.lat, p.lng) }))
          .sort((a, b) => a.d - b.d)
          .slice(0, 20)
          .map((x) => x.p)
      },
      () => (geoError = '無法取得定位，請改用逐層瀏覽'),
      { enableHighAccuracy: true, timeout: 10_000 },
    )
  }
</script>

<div class="search">
  <button class="primary" onclick={findNearby}>📍 附近的清運點</button>
  {#if geoError}<p class="muted">{geoError}</p>{/if}

  <div class="selectors">
    <select bind:value={area} onchange={() => { routeKeySel = ''; nearby = null }}>
      <option value="">選行政區</option>
      {#each areas as a}<option value={a}>{a}</option>{/each}
    </select>
    <select bind:value={routeKeySel} onchange={() => (nearby = null)} disabled={!area}>
      <option value="">選路線</option>
      {#each routes as r}<option value={r.key}>路線 {r.routeId}</option>{/each}
    </select>
    {#if routeKeySel}<input placeholder="過濾點名／里別" bind:value={keyword} />{/if}
  </div>

  <ul class="pointlist">
    {#each points as p (p.id)}
      <li>
        <div>
          <strong>{p.name}</strong>
          <span class="muted">{p.area}・{p.village}・{p.time ?? '無時刻'}・路線 {p.routeId}</span>
        </div>
        <button class="fav" class:on={favorites.has(p.id)} onclick={() => favorites.toggle(p.id)}>
          {favorites.has(p.id) ? '★' : '☆'}
        </button>
      </li>
    {/each}
  </ul>
</div>
```

### Task 9: 地圖頁

**Files:**
- Create: `src/views/MapView.svelte`
- Modify: `src/App.svelte`

**Interfaces:**
- Consumes: `favorites`、`pointById`、`routeKey`、`routePoints`、`gps`
- Produces: Leaflet 地圖——最愛點（★ marker）、其所屬路線的全部點（小圓點）、未過期車輛（🚛 marker，隨輪詢移動）。預設 fit 到最愛點範圍；無最愛時置中台南市區（22.99, 120.21, zoom 13）。

- [ ] **Step 1: 實作 `MapView.svelte`**

```svelte
<script lang="ts">
  import { onMount } from 'svelte'
  import L from 'leaflet'
  import 'leaflet/dist/leaflet.css'
  import { favorites } from '../lib/favorites.svelte'
  import { pointById, routeKey, routePoints } from '../lib/points'
  import { gps } from '../lib/gpsPoller.svelte'
  import { isStale } from '../lib/trucks'

  let el: HTMLDivElement
  let map: L.Map
  let truckLayer = L.layerGroup()

  const truckIcon = L.divIcon({ className: 'truck-icon', html: '🚛', iconSize: [28, 28] })
  const starIcon = L.divIcon({ className: 'star-icon', html: '⭐', iconSize: [24, 24] })

  onMount(() => {
    map = L.map(el, { zoomControl: false })
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '&copy; OpenStreetMap',
    }).addTo(map)
    truckLayer.addTo(map)

    const favs = favorites.ids.map((id) => pointById.get(id)).filter((p) => p != null)
    const bounds: L.LatLngTuple[] = []
    const routesDrawn = new Set<string>()
    for (const p of favs) {
      L.marker([p.lat, p.lng], { icon: starIcon }).bindPopup(`${p.name}<br>${p.time ?? ''}`).addTo(map)
      bounds.push([p.lat, p.lng])
      const k = routeKey(p)
      if (!routesDrawn.has(k)) {
        routesDrawn.add(k)
        for (const rp of routePoints(k))
          L.circleMarker([rp.lat, rp.lng], { radius: 3, color: '#ffc53d', weight: 1 }).addTo(map)
      }
    }
    if (bounds.length) map.fitBounds(bounds, { padding: [40, 40], maxZoom: 16 })
    else map.setView([22.99, 120.21], 13)

    return () => map.remove()
  })

  $effect(() => {
    const now = new Date()
    truckLayer.clearLayers()
    for (const t of gps.trucks) {
      if (isStale(t, now)) continue
      L.marker([t.lat, t.lng], { icon: truckIcon })
        .bindPopup(`${t.plate}<br>${t.caption}<br>${t.at.toLocaleTimeString('zh-TW')}`)
        .addTo(truckLayer)
    }
  })
</script>

<div class="map" bind:this={el}></div>

<style>
  .map { position: fixed; inset: 0 0 56px 0; }
  :global(.truck-icon), :global(.star-icon) { font-size: 22px; line-height: 1; background: none; border: none; }
</style>
```
